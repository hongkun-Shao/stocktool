<%*
/*
================================================================
  个股分析模板 — 从选股池联动创建深度分析笔记
  
  用法：
    1. 在 Obsidian 中运行此模板（Templater → 插入模板）
    2. 从选股池中选择要分析的股票
    3. 自动创建分析文档并双向链接到池文件
  
  联动机制：
    - 分析文件 frontmatter 含股票代码 → Dataview 可跨查询
    - 分析文件 → [[股票池/股票名]] 回链
    - 看板联动视图 → 显示分析状态
================================================================
*/

const poolFolder = "股票池";
const analysisFolder = "个股分析";

// 确保个股分析目录存在
if (!app.vault.getAbstractFileByPath(analysisFolder)) {
    await app.vault.createFolder(analysisFolder);
}

// 读取股票池目录，获取所有在池股票
const poolDir = app.vault.getAbstractFileByPath(poolFolder);
if (!poolDir || !poolDir.children) {
    throw new Error("股票池目录不存在或为空，请先运行归档脚本");
}

// 读取所有池文件的 frontmatter
const stocks = [];
for (const child of poolDir.children) {
    if (child.extension !== 'md') continue;
    try {
        const content = await app.vault.read(child);
        const codeMatch = content.match(/^代码: (.+)$/m);
        const nameMatch = content.match(/^名称: (.+)$/m);
        const industryMatch = content.match(/^行业: (.+)$/m);
        const divMatch = content.match(/^最新股息率: (.+)$/m);
        const statusMatch = content.match(/^状态: (.+)$/m);
        if (statusMatch && statusMatch[1].trim() === '在池') {
            stocks.push({
                name: nameMatch ? nameMatch[1].trim() : child.basename,
                code: codeMatch ? codeMatch[1].trim() : '',
                industry: industryMatch ? industryMatch[1].trim() : '',
                div: divMatch ? divMatch[1].trim() : '',
                file: child.path
            });
        }
    } catch (e) {
        console.log(`读取 ${child.name} 失败: ${e.message}`);
    }
}

if (stocks.length === 0) {
    throw new Error("股票池为空或没有在池股票");
}

// 让用户选择股票（建议器方式）
const stockNames = stocks.map(s => `${s.name} (${s.code})`);
let selectedLabel;
try {
    selectedLabel = await tp.system.suggester(
        items => items,
        stockNames,
        false,
        "选择要分析的股票："
    );
} catch (e) {
    throw new Error("用户取消了选择");
}

if (!selectedLabel) return;

// 解析选中的股票
const selected = stocks.find(s => `${s.name} (${s.code})` === selectedLabel);
if (!selected) return;

const today = moment().format("YYYY年MM月DD日");
const fileName = `${selected.name}.md`;
const filePath = `${analysisFolder}/${fileName}`;

// 检查是否已存在分析文件
if (app.vault.getAbstractFileByPath(filePath)) {
    const confirm = await tp.system.suggester(
        items => items,
        ['追加到已有分析', '取消'],
        false,
        `「${selected.name}」已有分析笔记，选择操作：`
    );
    if (confirm === '取消') return;
    // 如果选择追加，直接打开已有文件
    app.workspace.openLinkText(filePath, '', false);
    return;
}

const content = `---
股票名称: ${selected.name}
股票代码: ${selected.code}
行业: ${selected.industry}
股息率: ${selected.div}%
创建日期: ${today}
标签: 个股分析
状态: 追踪中
---

# ${selected.name} (${selected.code}) 分析笔记

> 数据来源：[[${selected.file}|选股池]] · 创建日期：${today}

---

## 商业模式

- **业务定位**：这家公司靠什么赚钱？
- **收入结构**：主营业务占比？是否有增长点？
- **客户/市场**：TO B 还是 TO C？市场空间多大？

---

## 竞争优势（护城河）

- [ ] 品牌溢价
- [ ] 技术/专利壁垒
- [ ] 规模/成本优势
- [ ] 网络效应/转换成本
- [ ] 特许经营/牌照优势

**综合评估**：

---

## 行业分析

- **行业阶段**：成长期 / 成熟期 / 衰退期
- **竞争格局**：集中度（CR3/CR5）、主要对手
- **周期性**：强周期 / 弱周期

---

## 财务健康

| 指标 | 当前值 | 评价 |
|------|--------|------|
| 股息率 | ${selected.div}% | |
| PE | | |
| ROE | | |
| 负债率 | | |

---

## 风险因素

- [ ] 行业政策风险
- [ ] 竞争加剧风险
- [ ] 客户集中风险
- [ ] 负债/流动性风险
- [ ] 股息可持续风险

---

## 估值与决策

- **合理估值区间**：
- **当前估值水平**：
- **投资策略**：关注 / 买入 / 持有 / 卖出
- **仓位建议**：

---

## 跟踪记录

### ${today}

创建初次分析笔记。

---

`;

await app.vault.create(filePath, content);
app.workspace.openLinkText(filePath, '', false);

console.log(`[个股分析] 已创建 ${filePath}`);
%>

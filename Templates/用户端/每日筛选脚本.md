<%*
/* 
================================================================
  A 股每日筛选入池脚本 (用户端 · 无 API 依赖)

  功能: 读取每日归档数据，筛选入池，评分排名，生成报告

  前置: 先运行「每日数据拉取脚本.md」从 OSS 下载当日归档文件
  输入: 股票历史记录/YYYY年MM月DD日.md (含全量行情+财务+历史EPS)
  输出:
    - 股票池/          → 累积池文件更新
    - 每日分析结果/     → 股息率排名.md + 选股变动.md

  与开发端的区别: 本脚本零 API 调用，所有数据来自归档 JSON
================================================================

const CONFIG_PATH = "选股配置.md";

async function loadConfig() {
    const defaults = {
        baseFolder: "股票历史记录",
        poolFolder: "股票池",
        reportFolder: "每日分析结果",
        targetDate: "",
        minDividend: 3.8,
        excludeNegativePE: true, excludeNegativeROE: true,
        maxDebtRatio: 60, minMarketCapYi: 20,
        scoringDividendWeight: 30, scoringPEWeight: 15, scoringPBWeight: 5,
        scoringDebtWeight: 10, scoringMarketCapWeight: 10,
        scoringROEWeight: 15, scoringProfitGrowthWeight: 10, scoringEPSWeight: 5,
        dividendNormCeil: 15, peNormMax: 20, pbNormMax: 5,
        roeNormMax: 30, profitGrowthNormRange: 50, epsNormMax: 5,
    };
    const file = app.vault.getAbstractFileByPath(CONFIG_PATH);
    if (!file) return defaults;
    const content = await app.vault.read(file);
    const fm = content.match(/^---\n([\s\S]*?)\n---/);
    if (!fm) return defaults;
    const vals = {};
    for (const line of fm[1].split('\n')) {
        const trimmed = line.trim();
        if (!trimmed || trimmed.startsWith('#')) continue;
        const m = trimmed.match(/^([\w-]+):\s*(.+?)(?:\s+#.*)?$/);
        if (!m) continue;
        let v = m[2].trim();
        if (v === 'true') v = true;
        else if (v === 'false') v = false;
        else if (/^-?\d+\.?\d*$/.test(v)) v = Number(v);
        vals[m[1]] = v;
    }
    const cfg = { ...defaults, ...vals };
    return {
        config: {
            baseFolder: cfg.baseFolder,
            minDividend: cfg.minDividend,
        },
        screenConfig: {
            excludeNegativePE: cfg.excludeNegativePE,
            excludeNegativeROE: cfg.excludeNegativeROE,
            maxDebtRatio: cfg.maxDebtRatio,
            minMarketCapYi: cfg.minMarketCapYi
        },
        scoring: {
            dividendWeight: cfg.scoringDividendWeight,
            peWeight: cfg.scoringPEWeight,
            pbWeight: cfg.scoringPBWeight,
            debtWeight: cfg.scoringDebtWeight,
            marketCapWeight: cfg.scoringMarketCapWeight,
            roeWeight: cfg.scoringROEWeight,
            profitGrowthWeight: cfg.scoringProfitGrowthWeight,
            epsWeight: cfg.scoringEPSWeight
        },
        normBounds: {
            dividendCeil: cfg.dividendNormCeil,
            peMax: cfg.peNormMax,
            pbMax: cfg.pbNormMax,
            roeMax: cfg.roeNormMax,
            profitGrowthRange: cfg.profitGrowthNormRange,
            epsMax: cfg.epsNormMax
        },
        folders: {
            history: cfg.baseFolder,
            pool: cfg.poolFolder,
            report: cfg.reportFolder
        },
        targetDate: cfg.targetDate,
    };
}

const cfg = await loadConfig();
const config = cfg.config;
const screenConfig = cfg.screenConfig;
const scoring = cfg.scoring;
const normBounds = cfg.normBounds;
const FOLDERS = cfg.folders;

// targetDate: 与归档脚本保持一致的日期，留空=今天
const targetDateRaw = cfg.targetDate ? String(cfg.targetDate) : '';
const useTarget = targetDateRaw && /^\d{8}$/.test(targetDateRaw);
let archiveMoment = useTarget ? moment(targetDateRaw, "YYYYMMDD") : moment();
if (useTarget && !archiveMoment.isValid()) archiveMoment = moment();
let dateStr = archiveMoment.format("YYYY年MM月DD日");

// ===== 初始化文件夹 =====
for (const dir of [FOLDERS.pool, FOLDERS.report]) {
    if (!app.vault.getAbstractFileByPath(dir))
        await app.vault.createFolder(dir);
}

function sanitizeFileName(name) {
    return name.replace(/[\\/:*?"<>|]/g, '');
}

function fmtCap(val) {  // val in 万元 (Tushare unit)
    const yi = val / 1e4;
    if (yi >= 100) return yi.toFixed(0) + " 亿";
    if (yi >= 1) return yi.toFixed(1) + " 亿";
    return (val / 1e4).toFixed(2) + " 亿";
}

// --- 评分归一化函数 ---
function normDividend(div) {
    const floor = config.minDividend || 3;
    const ceil = normBounds.dividendCeil;
    const range = ceil - floor;
    return Math.min(100, Math.max(0, (div - floor) / range * 100));
}
function normPE(peNum) {
    if (peNum == null) return 50;
    const max = normBounds.peMax;
    return Math.min(100, Math.max(0, (max - peNum) / max * 100));
}
function normPB(pbNum) {
    if (pbNum == null) return 50;
    const max = normBounds.pbMax;
    return Math.min(100, Math.max(0, (max - pbNum) / max * 100));
}
function normDebt(debtNum) {
    if (debtNum == null) return 50;
    return Math.min(100, Math.max(0, 100 - debtNum));
}
function normMarketCap(marketCapNum) {
    if (marketCapNum == null) return 0;
    return Math.min(100, Math.max(0, Math.log10(marketCapNum / 10 + 1) / Math.log10(501) * 100));
}
function normROE(roe) {
    if (roe === null) return 0;
    const max = normBounds.roeMax;
    return Math.min(100, Math.max(0, roe / max * 100));
}
function normProfitGrowth(g) {
    if (g == null) return 0;
    const range = normBounds.profitGrowthRange;
    return Math.min(100, Math.max(0, (g + range) / (range * 2) * 100));
}
function normEPS(eps) {
    if (eps === null) return 0;
    const max = normBounds.epsMax;
    return Math.min(100, Math.max(0, eps / max * 100));
}
function calcScore(stock, w) {
    const totalWeight = w.dividendWeight + w.peWeight + w.pbWeight + w.debtWeight + w.marketCapWeight + w.roeWeight + w.profitGrowthWeight + w.epsWeight;
    if (totalWeight === 0) return 0;
    let score = 0;
    score += normDividend(stock.div) * w.dividendWeight;
    score += normPE(stock.pe) * w.peWeight;
    score += normPB(stock.pb) * w.pbWeight;
    score += normDebt(stock.debtRatio) * w.debtWeight;
    score += normMarketCap(stock.marketCap) * w.marketCapWeight;
    score += normROE(stock.roe) * w.roeWeight;
    score += normProfitGrowth(stock.profitGrowth) * w.profitGrowthWeight;
    score += normEPS(stock.eps) * w.epsWeight;
    return Math.round(score / totalWeight * 10) / 10;
}

// =====================================================
//  步骤 1：读取当日归档数据
// =====================================================
console.log(`========================================`);
console.log(`[筛选入池] 日期: ${dateStr}`);
console.log(`[筛选条件] 股息率 >= ${config.minDividend}%${screenConfig.excludeNegativePE ? ' | 剔除PE为负' : ''}${screenConfig.excludeNegativeROE ? ' | 剔除ROE为负' : ''}${screenConfig.minMarketCapYi ? ` | 市值>=${screenConfig.minMarketCapYi}亿` : ''}`);
console.log(`========================================`);

let archivePath = `${FOLDERS.history}/${dateStr}.md`;
let archiveFile = app.vault.getAbstractFileByPath(archivePath);

// ===== 兜底方案：当日归档不存在时，自动使用本地最近有数据的日期 =====
if (!archiveFile) {
    console.log(`[筛选] 当日归档不存在: ${archivePath}`);
    console.log(`[筛选] 启用本地兜底，查找最近可用数据...`);
    const folder = app.vault.getAbstractFileByPath(FOLDERS.history);
    if (folder && folder.children) {
        const latestFile = folder.children
            .filter(f => f.extension === 'md' && /^\d{4}年\d{2}月\d{2}日\.md$/.test(f.name))
            .sort((a, b) => b.name.localeCompare(a.name))[0];
        if (latestFile) {
            dateStr = latestFile.name.replace('.md', '');
            archivePath = `${FOLDERS.history}/${latestFile.name}`;
            archiveFile = latestFile;
            console.log(`[筛选] 兜底使用本地最近数据: ${dateStr}`);
        }
    }
    if (!archiveFile) {
        throw new Error(`未找到任何归档文件，请先运行「用户端 → 每日数据拉取脚本」从 OSS 下载数据`);
    }
}

const archiveContent = await app.vault.read(archiveFile);
const jsonMatch = archiveContent.match(/```json\n([\s\S]*?)\n```/);
if (!jsonMatch) throw new Error("归档文件格式异常，找不到 JSON 数据块");

const allStocks = JSON.parse(jsonMatch[1]);
console.log(`[读取] 归档数据: ${allStocks.length} 只股票`);

// =====================================================
//  步骤 2：基于股息率初筛
// =====================================================
console.log(`\n[筛选] 开始筛选...`);

const prelimQualified = [];
for (const s of allStocks) {
    const divRaw = s.dv_ttm != null ? parseFloat(s.dv_ttm) : null;
    if (!s.ts_code || !s.name) continue;
    if (divRaw === null || isNaN(divRaw) || divRaw < config.minDividend) continue;
    prelimQualified.push(s);
}
console.log(`[筛选] 股息率≥${config.minDividend}%: ${prelimQualified.length} 只`);

// =====================================================
//  步骤 3：全量筛选 + 评分 + 更新池文件
// =====================================================
let qualifiedStocks = [];
let screenedOut = [];
for (const s of prelimQualified) {
    const name = s.name;
    const ts_code = s.ts_code || '';
    const code = ts_code.split('.')[0];

    const divRaw = s.dv_ttm != null ? parseFloat(s.dv_ttm) : null;
    const peRaw = s.pe_ttm != null ? Number(s.pe_ttm) : null;
    const pbNum = s.pb != null ? Number(s.pb) : null;
    const debtNum = s.debt_to_assets != null ? parseFloat(Number(s.debt_to_assets).toFixed(1)) : null;
    const marketCapRaw = s.total_mv ? Number(s.total_mv) : null;
    const marketCapNum = marketCapRaw ? marketCapRaw / 1e4 : null;
    const roe = s.roe != null ? parseFloat(Number(s.roe).toFixed(2)) : null;
    const profitGrowthNum = s.netprofit_yoy != null ? Number(s.netprofit_yoy) : null;
    const eps = s.eps != null ? parseFloat(Number(s.eps).toFixed(2)) : null;

    const priceNum = s.close != null ? Number(s.close) : null;
    const price = priceNum != null ? priceNum.toFixed(2) : "—";
    const pctRaw = s.pct_chg != null ? Number(s.pct_chg) : 0;
    const pct = pctRaw.toFixed(2);
    const pe = peRaw !== null && peRaw > 0 ? peRaw.toFixed(2) : (peRaw !== null && peRaw <= 0 ? peRaw.toFixed(2) : "—");
    const pb = pbNum !== null ? pbNum.toFixed(2) : "—";
    const debtRatio = debtNum !== null ? debtNum + "%" : "—";
    const marketCap = marketCapRaw ? fmtCap(marketCapRaw) : "—";
    const industry = s.industry || "—";
    const grossMargin = s.grossprofit_margin != null ? Number(s.grossprofit_margin).toFixed(1) + "%" : "—";
    const netMargin = s.netprofit_margin != null ? Number(s.netprofit_margin).toFixed(1) + "%" : "—";

    let reasons = [];
    if (screenConfig.excludeNegativePE && peRaw !== null && peRaw <= 0) {
        reasons.push(`PE为负(${peRaw.toFixed(2)})`);
    }
    if (screenConfig.excludeNegativeROE && roe !== null && roe < 0) {
        reasons.push(`ROE为负(${roe.toFixed(2)}%)`);
    }
    if (screenConfig.minMarketCapYi && marketCapNum !== null) {
        if (marketCapNum < screenConfig.minMarketCapYi) {
            reasons.push(`市值过低(${marketCapNum.toFixed(1)}亿)`);
        }
    }
    const isHighDebt = screenConfig.maxDebtRatio && s.debt_to_assets != null && s.debt_to_assets > screenConfig.maxDebtRatio;

    if (reasons.length > 0) {
        screenedOut.push({ name, code, div: divRaw, pe, debtRatio, marketCap, reasons: reasons.join("; ") });
        continue;
    }

    const stockScore = calcScore({ div: divRaw, pe: peRaw, pb: pbNum, debtRatio: debtNum, marketCap: marketCapNum, roe, profitGrowth: profitGrowthNum, eps }, scoring);
    qualifiedStocks.push({ name, code, ts_code, price, priceNum, pct, div: divRaw, industry, pe, peNum: peRaw, pb, pbNum, debtRatio, debtNum, marketCap, marketCapNum, isHighDebt, score: stockScore, roe, profitGrowth: profitGrowthNum ? profitGrowthNum.toFixed(1) + "%" : "—", profitGrowthNum, eps });

    // --- 累积股票池文件 ---
    const safeName = sanitizeFileName(name);
    const poolPath = `${FOLDERS.pool}/${safeName}.md`;
    const existingPool = app.vault.getAbstractFileByPath(poolPath);

    let entryDate = dateStr;
    let historyLines = [];
    if (existingPool) {
        const existingContent = await app.vault.read(existingPool);
        const fmMatch = existingContent.match(/^---\n[\s\S]*?\n---\n([\s\S]*)$/);
        if (fmMatch) {
            const bodyLines = fmMatch[1].split('\n').filter(l => l.trim());
            if (bodyLines.length > 0 && bodyLines[0].trim().startsWith('# ')) {
                historyLines = bodyLines.slice(1);
            } else {
                historyLines = bodyLines;
            }
        }
        const entryMatch = existingContent.match(/入池日期: (.+)/);
        if (entryMatch) entryDate = entryMatch[1].trim();
    }

    const chgSign = parseFloat(pct) >= 0 ? '+' : '';
    const newLine = `- 股息率: ${divRaw}% | PE: ${pe} | PB: ${pb} | 负债率: ${debtRatio} | 市值: ${marketCap} | 收盘: ${price} | 涨跌幅: ${chgSign}${pct}%`;
    const todayHeader = `## ${dateStr}`;
    const alreadyHasToday = historyLines.some(l => l.includes(todayHeader));

    let newBody;
    if (alreadyHasToday) {
        const dayIdx = historyLines.findIndex(l => l.includes(todayHeader));
        let endIdx = historyLines.findIndex((l, idx) => idx > dayIdx && l.startsWith('## '));
        if (endIdx === -1) endIdx = historyLines.length;
        newBody = [...historyLines.slice(0, dayIdx), todayHeader, newLine, ...historyLines.slice(endIdx)].join('\n');
    } else {
        newBody = [todayHeader, newLine, ...historyLines].join('\n');
    }

    const poolContent = `---
名称: ${name}
代码: ${code}
行业: ${industry}
入池日期: ${entryDate}
最新归档日期: ${dateStr}
最新股息率: ${divRaw}
最新市盈率: ${peRaw !== null ? peRaw : ''}
最新市净率: ${pbNum !== null ? pbNum : ''}
最新负债率: ${debtNum !== null ? debtNum : 0}
最新总市值: ${marketCap}
最新收盘价: ${priceNum !== null ? priceNum : ''}
最新涨跌幅: ${pct}
最新评分: ${stockScore}
最新ROE: ${roe !== null ? roe : ''}
最新毛利率: ${grossMargin}
最新净利率: ${netMargin}
最新净利同比增长: ${profitGrowthNum !== null ? profitGrowthNum : ''}
最新每股收益: ${eps !== null ? eps : ''}
状态: 在池
---

# ${name} (${code})

${newBody}
`;

    if (existingPool) {
        await app.vault.modify(existingPool, poolContent);
    } else {
        await app.vault.create(poolPath, poolContent);
    }
    console.log(`[池] ${name} | ${industry} | 股息率 ${divRaw}% | PE ${pe} | ROE ${roe !== null ? roe + '%' : '—'} | 评分 ${stockScore}${isHighDebt ? ' ⚠️高负债' : ''}`);
}

// =====================================================
//  步骤 4：入出池变动报告
// =====================================================
const reportPath = `${FOLDERS.report}/${dateStr}`;
if (!app.vault.getAbstractFileByPath(reportPath))
    await app.vault.createFolder(reportPath);

const poolDir = app.vault.getAbstractFileByPath(FOLDERS.pool);
let exitedStocks = [];
let updatedCodes = new Set(qualifiedStocks.map(s => s.code));
if (poolDir && poolDir.children) {
    for (let child of poolDir.children) {
        if (child.extension !== 'md') continue;
        try {
            const content = await app.vault.read(child);
            const statusMatch = content.match(/^状态: (.+)$/m);
            if (statusMatch && statusMatch[1].trim() === '在池') {
                const codeMatch = content.match(/^代码: (.+)$/m);
                const stockCode = codeMatch ? codeMatch[1].trim() : '';
                if (stockCode && !updatedCodes.has(stockCode)) {
                    const nameMatch = content.match(/^名称: (.+)$/m);
                    const lastDateMatch = content.match(/^最新归档日期: (.+)$/m);
                    const lastDivMatch = content.match(/^最新股息率: (.+)$/m);
                    const lastPeMatch = content.match(/^最新市盈率: (.+)$/m);
                    exitedStocks.push({
                        name: nameMatch ? nameMatch[1].trim() : '未知',
                        code: stockCode,
                        lastDate: lastDateMatch ? lastDateMatch[1].trim() : '未知',
                        lastDiv: lastDivMatch ? lastDivMatch[1].trim() : '?',
                        lastPe: lastPeMatch ? lastPeMatch[1].trim() : '?'
                    });
                }
            }
        } catch (e) {
            console.log(`[警告] 读取 ${child.name} 失败: ${e.message}`);
        }
    }
}

for (let s of exitedStocks) {
    const path = `${FOLDERS.pool}/${sanitizeFileName(s.name)}.md`;
    const f = app.vault.getAbstractFileByPath(path);
    if (!f) continue;
    try {
        const content = await app.vault.read(f);
        const updated = content.replace(/^状态: 在池$/m, '状态: 已退出');
        if (updated !== content) { await app.vault.modify(f, updated); console.log(`[退出] ${s.name} 状态已更新`); }
    } catch (e) { console.log(`[退出] 更新 ${s.name} 失败: ${e.message}`); }
}

const changePath = `${reportPath}/选股变动.md`;
let changeContent = `# ${dateStr} 选股变动

> 筛选条件：股息率 ≥ ${config.minDividend}%${screenConfig.excludeNegativePE ? ' | 剔除PE为负' : ''}${screenConfig.excludeNegativeROE ? ' | 剔除ROE为负' : ''}${screenConfig.maxDebtRatio ? ` | 负债率≤${screenConfig.maxDebtRatio}%` : ''}${screenConfig.minMarketCapYi ? ` | 市值≥${screenConfig.minMarketCapYi}亿` : ''}

`;
if (qualifiedStocks.length > 0) {
    changeContent += `## ✅ 新入池 (${qualifiedStocks.length} 只)\n\n`;
    changeContent += `| 评分 | 名称 | 代码 | 行业 | 股息率 | PE | ROE | 负债率 | 总市值 |\n|:---:|:---|:---:|:---|:---:|:---:|:---:|:---:|:---:|\n`;
    for (let s of qualifiedStocks) {
        const warnFlag = s.isHighDebt ? ' ⚠️' : '';
        const roeDisplay = s.roe !== null ? s.roe + '%' : '—';
        changeContent += `| ${s.score} | ${s.name}${warnFlag} | ${s.code} | ${s.industry} | ${s.div}% | ${s.pe} | ${roeDisplay} | ${s.debtRatio} | ${s.marketCap} |\n`;
    }
    changeContent += '\n';
}
if (screenedOut.length > 0) {
    changeContent += `## ❌ 被筛除 (${screenedOut.length} 只)\n\n| 名称 | 代码 | 股息率 | PE | 负债率 | 原因 |\n|:---|:---:|:---:|:---:|:---:|:---|\n`;
    for (let s of screenedOut) changeContent += `| ${s.name} | ${s.code} | ${s.div}% | ${s.pe} | ${s.debtRatio} | ${s.reasons} |\n`;
    changeContent += '\n';
}
if (exitedStocks.length > 0) {
    changeContent += `## ⏳ 暂未达标 (${exitedStocks.length} 只)\n\n| 名称 | 代码 | 前日股息率 | 前日PE |\n|:---|:---:|:---:|:---:|\n`;
    for (let s of exitedStocks) changeContent += `| ${s.name} | ${s.code} | ${s.lastDiv}% | ${s.lastPe} |\n`;
    changeContent += '\n';
}
changeContent += `---\n_生成时间: ${moment().format('HH:mm:ss')}_\n`;

const existingChange = app.vault.getAbstractFileByPath(changePath);
if (existingChange) { await app.vault.modify(existingChange, changeContent); }
else { await app.vault.create(changePath, changeContent); }
console.log(`[变动报告] 已生成 ${changePath}`);

// =====================================================
//  步骤 5：生成汇总排名表
// =====================================================
const tablePath = `${reportPath}/股息率排名.md`;
const rankedStocks = [...qualifiedStocks].sort((a, b) => b.score - a.score);
let tableContent = `# ${dateStr} 高息股排名

> 筛选条件：股息率 ≥ ${config.minDividend}%${screenConfig.excludeNegativePE ? ' · 剔除PE为负' : ''}${screenConfig.excludeNegativeROE ? ' · 剔除ROE为负' : ''}${screenConfig.maxDebtRatio ? ` · 负债率≤${screenConfig.maxDebtRatio}%` : ''}${screenConfig.minMarketCapYi ? ` · 市值≥${screenConfig.minMarketCapYi}亿` : ''} | 数据来源：OSS归档数据 | 共 ${qualifiedStocks.length} 只 | 按综合评分降序

| 排名 | 评分 | 名称 | 代码 | 行业 | 股息率 | 市盈率 | 市净率 | 负债率 | ROE | 总市值 | 收盘价 | 涨跌幅 |
|:---:|:---:|:---|:---:|:---|:---:|:---:|:---:|:---:|:---:|:---:|:---:|:---:|
`;
rankedStocks.forEach((s, i) => {
    const rank = i + 1;
    const chgSign = parseFloat(s.pct) >= 0 ? '+' : '';
    const roeDisplay = s.roe !== null ? s.roe + '%' : '—';
    tableContent += `| ${rank} | ${s.score} | ${s.name} | ${s.code} | ${s.industry} | ${s.div}% | ${s.pe} | ${s.pb} | ${s.debtRatio} | ${roeDisplay} | ${s.marketCap} | ${s.price} 元 | ${chgSign}${s.pct}% |\n`;
});

const existingTable = app.vault.getAbstractFileByPath(tablePath);
if (existingTable) { await app.vault.modify(existingTable, tableContent); }
else { await app.vault.create(tablePath, tableContent); }
console.log(`[表格] 已生成 ${tablePath}`);

// =====================================================
//  步骤 6：总结
// =====================================================
console.log(`\n========================================`);
console.log(`筛选入池完成`);
console.log(`  读取归档: ${allStocks.length} 只`);
console.log(`  入池: ${qualifiedStocks.length} 只`);
console.log(`  被筛除: ${screenedOut.length} 只`);
console.log(`  暂未达标: ${exitedStocks.length} 只`);
console.log(`  汇总表格: 股息率排名.md`);
console.log(`  变动报告: 选股变动.md`);
console.log(`========================================`);

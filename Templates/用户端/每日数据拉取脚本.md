<%*
/*
================================================================
  每日数据拉取脚本 (用户端)

  功能: 从 OSS/CDN 下载当日归档文件到本地
  前置: 开发端已将归档文件上传到 OSS
  输出: 股票历史记录/YYYY年MM月DD日.md

  无需 Tushare token，纯 HTTP GET 下载
================================================================
*/

const CONFIG_PATH = "选股配置.md";

async function loadConfig() {
    const defaults = {
        baseFolder: "股票历史记录",
        targetDate: "",
        archiveRetryDays: 5,
        ossUrl: "",  // OSS 基础 URL，不含文件名
    };
    const file = app.vault.getAbstractFileByPath(CONFIG_PATH);
    if (!file) return defaults;
    const content = await app.vault.read(file);
    const fm = content.match(/^---\n([\s\S]*?)\n---/);
    if (!fm) return defaults;
    const vals = {};
    for (const line of fm[1].split('\n')) {
        const trimmed = line.trim();
        if (!trimmed || trimmed.startsWith('#')) continue;
        const m = trimmed.match(/^([\w-]+):\s*(.+?)(?:\s+#.*)?$/);
        if (!m) continue;
        let v = m[2].trim();
        if (v === 'true') v = true;
        else if (v === 'false') v = false;
        else if (/^-?\d+\.?\d*$/.test(v)) v = Number(v);
        vals[m[1]] = v;
    }
    return { ...defaults, ...vals };
}

const cfg = await loadConfig();
const BASE_FOLDER = cfg.baseFolder;
const RETRY_DAYS = cfg.archiveRetryDays || 5;
const OSS_URL = cfg.ossUrl || "https://stocktool.oss-cn-hangzhou.aliyuncs.com";

const targetDateRaw = cfg.targetDate ? String(cfg.targetDate) : '';
const useTarget = targetDateRaw && /^\d{8}$/.test(targetDateRaw);
const archiveMoment = useTarget ? moment(targetDateRaw, "YYYYMMDD") : moment();
const dateStr = archiveMoment.format("YYYY年MM月DD日");

const httpGet = async (params) => {
    const fn = typeof requestUrl !== 'undefined' ? requestUrl : null;
    if (!fn) throw new Error("requestUrl 不可用，请确认 Templater 插件已安装并启用");
    const resp = await fn({ ...params, throw: false });
    if (resp.status < 200 || resp.status >= 300) {
        throw new Error(`HTTP ${resp.status}`);
    }
    return resp;
};

// ===== 初始化文件夹 =====
if (!app.vault.getAbstractFileByPath(BASE_FOLDER)) {
    await app.vault.createFolder(BASE_FOLDER);
}

console.log(`========================================`);
console.log(`[数据拉取] 目标日期: ${dateStr}`);
console.log(`[数据拉取] OSS 地址: ${OSS_URL}`);
console.log(`========================================`);

const isTransientError = (e) => {
    const msg = e.message || '';
    return /ERR_|net::|fetch|network|timeout|abort|5\d\d/i.test(msg);
};

let downloaded = false;
for (let retry = 0; retry < RETRY_DAYS && !downloaded; retry++) {
    const tryMoment = archiveMoment.clone().subtract(retry, 'days');
    const tryDateStr = tryMoment.format("YYYY年MM月DD日");
    const fileName = encodeURIComponent(tryDateStr + '.md');
    const url = `${OSS_URL}/${fileName}`;

    for (let attempt = 0; attempt < 2; attempt++) {
        console.log(`[数据拉取] 尝试:${tryDateStr}${attempt > 0 ? ` (重试${attempt})` : ''}`);
        console.log(`  URL: ${url}`);
        try {
            const resp = await httpGet({ url, method: 'GET' });
            if (!resp || !resp.text) {
                console.log(`  响应为空，继续重试...`);
                continue;
            }
            const content = resp.text.trim();
            if (!content) {
                console.log(`  内容为空，继续重试...`);
                continue;
            }
            const jsonMatch = content.match(/```json\s*([\s\S]*?)\s*```/);
            if (!jsonMatch) {
                console.log(`  文件不包含 JSON 数据块，继续重试...`);
                continue;
            }

            const filePath = `${BASE_FOLDER}/${tryDateStr}.md`;
            const existing = app.vault.getAbstractFileByPath(filePath);
            if (existing) {
                await app.vault.modify(existing, content);
            } else {
                await app.vault.create(filePath, content);
            }

            console.log(`[数据拉取] 成功 → ${filePath} (${content.length} 字节)`);
            downloaded = true;
            break;
        } catch (e) {
            if (!isTransientError(e)) {
                console.log(`  不可恢复错误，跳过该日期: ${e.message}`);
                break;
            }
            if (attempt < 1) {
                console.log(`  临时错误，重试同一天: ${e.message}`);
            } else {
                console.log(`  该日期重试耗尽: ${e.message}`);
            }
        }
    }
}

// ===== 兜底方案：OSS 全部失败时，使用本地最近有数据的日期 =====
if (!downloaded) {
    console.log(`[数据拉取] OSS 下载全部失败，启用本地兜底...`);
    const folder = app.vault.getAbstractFileByPath(BASE_FOLDER);
    if (folder && folder.children) {
        const latestFile = folder.children
            .filter(f => f.extension === 'md' && /^\d{4}年\d{2}月\d{2}日\.md$/.test(f.name))
            .sort((a, b) => b.name.localeCompare(a.name))[0];
        if (latestFile) {
            const latestDateStr = latestFile.name.replace('.md', '');
            const sourcePath = `${BASE_FOLDER}/${latestFile.name}`;
            const targetPath = `${BASE_FOLDER}/${dateStr}.md`;
            const content = await app.vault.read(latestFile);
            const existing = app.vault.getAbstractFileByPath(targetPath);
            if (existing) {
                await app.vault.modify(existing, content);
            } else {
                await app.vault.create(targetPath, content);
            }
            console.log(`[数据拉取] 兜底成功 → 本地最近数据: ${latestDateStr} → 复制为 ${dateStr}.md`);
            downloaded = true;
        }
    }
}

if (!downloaded) {
    throw new Error(`数据拉取失败：OSS 向前重试 ${RETRY_DAYS} 天均无可用归档文件，本地也无历史数据`);
}

console.log(`[数据拉取] 完成`);
%>

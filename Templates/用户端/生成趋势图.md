<%*
/*
================================================================
  生成趋势图 (用户端)

  功能: 读取选股池文件的历史记录，解析指定指标，生成 Mermaid 趋势图
  输出: 每日分析结果/YYYY年MM月DD日/趋势图.md (可追加多只股票)
================================================================
*/

const poolFolder = "股票池";
const reportFolder = "每日分析结果";

// 可用指标定义
const metrics = [
    { label: "股息率 (%)",     regex: /股息率:\s*([\d.]+)%/ },
    { label: "市盈率 (PE)",    regex: /PE:\s*([\d.]+)/ },
    { label: "市净率 (PB)",    regex: /PB:\s*([\d.]+)/ },
    { label: "负债率 (%)",     regex: /负债率:\s*([\d.]+)%/ },
    { label: "收盘价 (元)",    regex: /收盘:\s*([\d.]+)/ },
    { label: "涨跌幅 (%)",     regex: /涨跌幅:\s*([+-]?[\d.]+)%/ },
    { label: "市值 (亿)",      regex: /市值:\s*([\d.]+)\s*亿/ },
];

// ===== 步骤 1：读取股票池 =====
const poolDir = app.vault.getAbstractFileByPath(poolFolder);
if (!poolDir || !poolDir.children) {
    throw new Error("股票池目录不存在或为空，请先运行筛选脚本");
}

const stocks = [];
for (const child of poolDir.children) {
    if (child.extension !== 'md') continue;
    try {
        const fc = await app.vault.read(child);
        const sm = fc.match(/^状态: (.+)$/m);
        if (sm && sm[1].trim() === '在池') {
            const nm = fc.match(/^名称: (.+)$/m);
            stocks.push({
                name: nm ? nm[1].trim() : child.basename,
                file: child
            });
        }
    } catch (e) {}
}
if (stocks.length === 0) throw new Error("股票池中没有在池股票");

// ===== 步骤 2：选股票 =====
const selectedStock = await tp.system.suggester(
    s => s.name, stocks, false, "选择股票："
);
if (!selectedStock) return;

const poolContent = await app.vault.read(selectedStock.file);

// ===== 步骤 3：选指标 =====
const selectedMetric = await tp.system.suggester(
    m => m.label, metrics, false, "选择指标："
);
if (!selectedMetric) return;

// ===== 步骤 4：解析历史记录 =====
const bodyMatch = poolContent.match(/^---\n[\s\S]*?\n---\n([\s\S]*)$/);
if (!bodyMatch) throw new Error("无法解析池文件格式");

const sections = bodyMatch[1].split(/^## /m).filter(s => s.trim());
const dataPoints = [];

for (const section of sections) {
    const lines = section.split('\n');
    const header = lines[0].trim();
    const dm = header.match(/^(\d{4})年(\d{2})月(\d{2})日$/);
    if (!dm) continue;
    const dateShort = `${dm[2]}/${dm[3]}`;

    for (let i = 1; i < lines.length; i++) {
        const m = lines[i].match(selectedMetric.regex);
        if (m) {
            dataPoints.push({ dateShort, value: parseFloat(m[1]) });
            break;
        }
    }
}

if (dataPoints.length < 2) {
    throw new Error(`${selectedStock.name} 只有 ${dataPoints.length} 个数据点，不足以生成趋势图（至少需 2 个）`);
}

dataPoints.reverse();  // 旧→新顺序

// ===== 步骤 5：生成 Mermaid =====
const allValues = dataPoints.map(d => d.value);
const maxRaw = Math.max(...allValues);
const minRaw = Math.min(...allValues);
const margin = (maxRaw - minRaw) * 0.15 || 1;
const yMin = +(Math.max(0, minRaw - margin)).toFixed(1);
const yMax = +(maxRaw + margin).toFixed(1);

// 限制 x 轴标签数量
const maxLabels = 15;
const step = dataPoints.length > maxLabels ? Math.ceil(dataPoints.length / maxLabels) : 1;
const xLabels = dataPoints.map((d, i) => i % step === 0 ? `"${d.dateShort}"` : `""`);
const lineValues = dataPoints.map(d => d.value.toFixed(2));

const count = dataPoints.length;
const firstDate = dataPoints[0].dateShort.replace('/', '月') + '日';
const lastDate = dataPoints[count-1].dateShort.replace('/', '月') + '日';

const chartSection = `
## ${selectedStock.name} — ${selectedMetric.label}

> ${count} 个交易日 · ${firstDate} ~ ${lastDate} · 最高 ${maxRaw.toFixed(2)} · 最低 ${minRaw.toFixed(2)}

\`\`\`mermaid
xychart-beta
    title "${selectedStock.name} ${selectedMetric.label}"
    x-axis [${xLabels.join(', ')}]
    y-axis "${selectedMetric.label}" ${yMin} --> ${yMax}
    line [${lineValues.join(', ')}]
\`\`\`
`;

// ===== 步骤 6：写入文件 =====
const today = moment().format("YYYY年MM月DD日");
const reportDir = `${reportFolder}/${today}`;
if (!app.vault.getAbstractFileByPath(reportDir)) {
    await app.vault.createFolder(reportDir);
}

const chartPath = `${reportDir}/趋势图.md`;
const existingFile = app.vault.getAbstractFileByPath(chartPath);

if (existingFile) {
    let existingContent = await app.vault.read(existingFile);
    // 替换同股票同指标的旧图表，否则追加
    const marker = `## ${selectedStock.name} — ${selectedMetric.label}`;
    if (existingContent.includes(marker)) {
        const idx = existingContent.indexOf(marker);
        const nextIdx = existingContent.indexOf('\n## ', idx + 1);
        if (nextIdx > 0) {
            existingContent = existingContent.slice(0, idx) + chartSection.trim() + '\n' + existingContent.slice(nextIdx + 1);
        } else {
            existingContent = existingContent.slice(0, idx) + chartSection.trim();
        }
        await app.vault.modify(existingFile, existingContent);
        console.log(`[趋势图] 已更新 ${selectedStock.name} ${selectedMetric.label}`);
    } else {
        await app.vault.modify(existingFile, existingContent + '\n' + chartSection);
        console.log(`[趋势图] 已追加 ${selectedStock.name} ${selectedMetric.label}`);
    }
} else {
    const header = `# ${today} 趋势图\n\n> 数据来源：选股池 · 生成时间：${moment().format('HH:mm:ss')}`;
    await app.vault.create(chartPath, header + chartSection);
    console.log(`[趋势图] 已创建 ${chartPath}`);
}

console.log(`[趋势图] 完成: ${selectedStock.name} ${selectedMetric.label} (${count} 个数据点)`);
%>
